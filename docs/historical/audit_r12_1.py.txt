class Value:

    def get(self):return self.x

class Geometry(InstallationGeometryMixin,ShadowGeometryMixin,ZoneToolsMixin):
    def draw_grid(self):pass
    def _update_zone_combo(self):pass
    def _update_zone_entries_from_active(self):pass
    def _clean_deleted_panels_from_strings(self):
        self.strings={s:[c for c in coords if c in self.panels] for s,coords in self.strings.items()}

def geometry():
    g=Geometry();g.px_per_mm=.1;g.panel_width_mm=1000.;g.panel_height_mm=1700.
    g.roof_zones=[{'id':1,'x1':0,'x2':500,'y1':0,'y2':510},
